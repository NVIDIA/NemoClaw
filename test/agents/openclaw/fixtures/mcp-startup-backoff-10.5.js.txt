var McpStartupBackoffError = class extends Error {
	constructor(state, reportFailure, retryAfterMs) {
		super(`${state.message}; server unavailable; retry after ${new Date(retryAfterMs).toISOString()}. Check server reachability or reload MCP after fixing it.`);
		this.reportFailure = reportFailure;
		this.retryAfterMs = retryAfterMs;
	}
};
function resetMcpStartupBackoff(key) {

}
